const int TRIG = 3;       // 超声波 Trig：3 号的 S
const int ECHO = 2;       // 超声波 Echo：2 号的 S
 
int readDistance() {      // 测一次距离，交回厘米数（没测到交回 999）
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);       // 叫一声：10 微秒
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  unsigned long us = pulseIn(ECHO, HIGH, 30000);   // 等回声，最多 30 毫秒
  if (us == 0) {
    return 999;                   // 没收到回声：当成很远
  }
  return us / 58;                 // 来回 1 厘米约 58 微秒
}

★★ 三档：if、else if、else

loop：三档14 行
void loop() {
  int cm = readDistance();
  digitalWrite(RED, LOW);        // 先全灭
  digitalWrite(YELLOW, LOW);
  digitalWrite(GREEN, LOW);
  if (cm < 20) {                 // 近：红
    digitalWrite(RED, HIGH);
  } else if (cm < 50) {          // 中：黄
    digitalWrite(YELLOW, HIGH);
  } else {                       // 远：绿
    digitalWrite(GREEN, HIGH);
  }
  delay(100);
}
先加 YELLOW = 9、GREEN = 8 和它们的 pinMode；每圈先全灭，再亮该亮的一个。

（★★ 的组做的时候开着这一页。）

「先在上面加黄灯 9、绿灯 8 的名字，setup 里加 pinMode。」

「loop 里，每一圈先把三个灯全灭，再决定亮哪一个——这样不会两个灯一起亮。」

「判断从上往下问：小于 20 吗？是，红灯，后面的不再问；否则，如果小于 50，黄灯；都不是，绿灯。else if 就是『否则，如果』，中间要有空格，写成 elseif 是红字。」

「想一想：为什么第二个只写 cm < 50，不用写 cm >= 20？——能走到这里，前面已经答了『不小于 20』。」
