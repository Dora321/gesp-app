起步全码 · 第①-2 步：回调（L20 老本事）
def on_message(topic, msg):      # topic 参数必须留（库固定传两个）
    cmd = msg.decode()           # 音箱认出的指令词
    if cmd == "on":
        led.value(1)
    elif cmd == "off":
        led.value(0)
    elif cmd == "alarm":         # 警报模式：灯 + 蜂鸣一起动
        led.value(1)
        buzzer.value(1); time.sleep(1); buzzer.value(0)
